// ==================== YOUTUBE SEARCH ====================
gmd({
    pattern: 'yts',
    alias: ['ytsearch', 'youtubesearch'],
    desc: 'Search for YouTube Videos',
    category: 'search',
    react: '🔍',
    filename: __filename,
    use: '<search query>'
}, async (sock, message, messageData, { from, args, reply }) => {
    if (!args[0]) return reply('Please provide a search query!');
    const query = args.join(' ');
    try {
        const searchResults = await yts(query);
        if (!searchResults.videos.length) return reply('No videos found for the given query.');
        
        const firstVideo = searchResults.videos[0];
        let resultText = '*YouTube Search Results:*\n\n';

            resultText += `   Channel: ${video.author.name}\n`;
            resultText += `   ⏱️ Duration: ${video.duration.timestamp}\n`;
            resultText += `   👁️ Views: ${formatNumber(video.views)}\n`;
            resultText += `   🔗 Link: ${video.url}\n`;
            resultText += `   📅 Uploaded: ${video.ago}\n\n`;
        });

        const messageContent = {
            image: { url: firstVideo.thumbnail },
            caption: resultText,
            contextInfo: {
                mentionedJid: [messageData.sender],
                forwardingScore: 5,
                isForwarded: true,
                forwardedNewsletterMessageInfo: {
                    newsletterJid: '120363420122180789@newsletter',
                    newsletterName: 'JawadTechX',
                    serverMessageId: 143
                }
            }
        };
        await sock.sendMessage(from, messageContent, { quoted: message });
    } catch (error) {
        console.log('Error in YouTube search:', error);
        reply('❌ An error occurred while searching YouTube. Please try again later.');
    }
});

function formatNumber(num) {
    if (num >= 1000000) return (num / 1000000).toFixed(1) + 'M';
    else {
        if (num >= 1000) return (num / 1000).toFixed(1) + 'K';
    }
    return num.toString();
}

// ==================== PINTEREST IMAGE SEARCH ====================
gmd({
    pattern: 'img2',
    alias: ['image2', 'pint', 'pinterest'],
    desc: 'Search for Media from Pinterest.',
    react: '🖼️',
    category: 'search',
    filename: __filename
}, async (sock, message, messageData, { from, quoted, args, q, reply }) => {
    try {
        if (!q) return reply('Please provide a search query.');
        const apiUrl = global.api + '/search/pinterest?apikey=' + global.myName + '&query=' + encodeURIComponent(q);
        const response = await axios.get(apiUrl);
        const imageData = response.data.result;
        
        if (!imageData || imageData.length === 0) return reply('No images found for your search.');
        
        const imageUrls = imageData.slice(0, 6);
        const downloadableImages = [];
        
        for (const url of imageUrls) {
            try {
                const imgResponse = await axios.get(url, { responseType: 'arraybuffer' });
                if (imgResponse.status === 200) {
                    downloadableImages.push({
                        buffer: Buffer.from(imgResponse.data),
                        directLink: url
                    });
                }
            } catch (err) {
                console.log('Failed to fetch image: ' + url);
            }
        }
        
        if (downloadableImages.length === 0) return reply('All image downloads failed.');
        
        const cards = await Promise.all(downloadableImages.map(async (img, index) => ({
            header: {
                title: 'Image ' + (index + 1),
                hasMediaAttachment: true,
                imageMessage: (await generateWAMessageContent(
                    { image: img.buffer },
                    { upload: sock.waUploadToServer }
                )).imageMessage
            },
            body: { text: 'Search: ' + q },
            footer: { text: '🔁 Swipe to see more' },
            nativeFlowMessage: {
                buttons: [{
                    name: 'cta_url',
                    buttonParamsJson: JSON.stringify({
                        display_text: '🌐 View Image',
                        url: img.directLink
                    })
                }]
            }
        })));

        const interactiveMsg = generateWAMessageFromContent(from, {
            viewOnceMessage: {
                message: {
                    messageContextInfo: {
                        deviceListMetadata: {},
                        deviceListMetadataVersion: 2
                    },
                    interactiveMessage: {
                        body: { text: 'Search: ' + q },
                        footer: { text: '📦 Showing ' + downloadableImages.length + ' images' },
                        carouselMessage: { cards: cards }
                    }
                }
            }
        }, { quoted: message });
        
        await sock.relayMessage(from, interactiveMsg.message, { messageId: interactiveMsg.key.id });
    } catch (error) {
        console.log('Error in image command:', error);
        reply('❌ An error occurred while fetching images.');
    }
});
