    pattern: 'play',
    alias: ['ytmp3', 'yta'],
    desc: 'Download YouTube songs',
    category: 'downloader',
    react: '🎵',
    filename: __filename
}, async (sock, message, messageData, { from, args, q, reply }) => {
    try {
        if (!q) {
            return reply('🎶 Please provide song name!\n\nExample: .play Moye Moye');
        }

        const { videos } = await yts(q);
        if (!videos || videos.length === 0) {
            return reply('❌ No results found!');
        }

        const video = videos[0];
        const apiUrl = 'https://apis-keith.vercel.app/download/dlmp3?url=' + encodeURIComponent(video.url);
        const response = await axios.get(apiUrl);
        const data = response.data;

        if (!data?.status || !data?.result?.data?.downloadUrl) {
            return reply('❌ Download failed! Try again later.');
        }

        const downloadUrl = data.result.data.downloadUrl;
        const title = data.result.data.title || 'audio';

        await sock.sendMessage(from, {
            audio: { url: downloadUrl },
            mimetype: 'audio/mpeg',
            fileName: title + '.mp3'
        }, { quoted: message });

        await sock.sendMessage(from, {
            react: { text: '✅', key: messageData.key }
        });
    } catch (error) {
        console.error('Error in play command:', error);
        reply('❌ Error occurred, try again later!');
        await sock.sendMessage(from, {
            react: { text: '❌', key: messageData.key }
        });
    }
});

// ==================== GITHUB REPO CLONE ====================
gmd({
    pattern: 'gitclone',
    desc: 'Clone/Download GitHub Repositories',
    category: 'downloader',
    filename: __filename
}, async (sock, message, messageData, { 
    from, quoted, body, isCmd, command, args, q, isGroup, sender, 
    senderNumber, botNumber2, botNumber, pushname, isMe, isOwner, 
    groupMetadata, groupName, participants, groupAdmins, isBotAdmins, 
    isAdmins, reply 
}) => {
    try {
        if (!q) {
            return reply('Please provide a GitHub repository URL, e.g.,\n' + prefix + 'gitclone https://github.com/JawadYT36/KHAN-MD');
        }

        const cleanUrl = q.trim();
        if (!cleanUrl.includes('github.com')) {
            return reply('The provided URL does not appear to be a valid GitHub repository link.');
        }

        const repoPattern = /(?:https:\/\/|git@)github\.com[\/:]([^\/:]+)\/([^\/:\.]+)(?:\.git)?/i;
        const match = cleanUrl.match(repoPattern);

        if (!match) {
            return reply('Is that a valid GitHub repo link?!');
        }

        const [, owner, repo] = match;
        const zipUrl = 'https://api.github.com/repos/' + owner + '/' + repo + '/zipball';
        const response = await fetch(zipUrl, { method: 'HEAD' });

        if (!response.ok) {
            return reply('Failed to fetch repository details. Make sure the URL is correct and the repository is public.');
        }

        let fileName = response.headers.get('content-disposition');
        let finalName = fileName ? fileName.match(/attachment; filename=(.*)/)[1] : repo + '.zip';

        await sock.sendMessage(from, {
            document: { url: zipUrl },
            mimetype: 'application/zip',
            fileName: finalName
        }, { quoted: message });
    } catch (error) {
        console.error(error);
        reply('An error occurred: ' + error.message);
    }
});

// ==================== APK DOWNLOADER ====================
gmd({
    pattern: 'apk',
    alias: ['apkdl'],
    desc: 'Download Android Apps',
    use: '.apk <app_name>',
    react: '📥',
    category: 'downloader',
    filename: __filename
}, async (sock, message, messageData, { from, quoted, body, q, reply }) => {
    const appName = q.trim();
    if (!appName) {
        return reply('Please provide an app name for the link ❗');
    }

    reply('_Downloading ' + appName + '_');
    try {
        const apkData = await GiftedApkDl(appName);
        const buffer = await getBuffer(apkData.downloadUrl);

        if (!buffer || !apkData.appname) {
            await messageData.react('❌');
        }

        await sock.sendMessage(from, {
            document: buffer,
            caption: 'Downloaded Successfully ✅*\n> *© ᴘσωєʀє∂ ву ᴊᴀᴡᴀᴅ-ᴍᴅ⎯꯭̽🚩°*',
            mimetype: 'application/vnd.android.package-archive',
            fileName: apkData.appname + '.apk'
        }, { quoted: message });

        await messageData.react('✅');
    } catch (error) {
        console.log(error);
        await messageData.react('❌');
        reply('Error: ' + error.message);
    }
});
