  try {
    const antiCallMsg = document.getElementById('suc-antiCallMsg').value.trim();
    const payload = {
      botMode:         document.getElementById('suc-privateMode').checked ? 'private' : 'public',
      publicWelcome:   document.getElementById('suc-publicWelcome').checked,

    };

      method: 'POST',
      headers: { 'Content-Type': 'application/json' },

    if (btn) { btn.disabled = false; btn.textContent = '💾 Save My Settings'; }
  }
}

async function cStart() {
  const errEl = document.getElementById('c-err');
  errEl.style.display = 'none';
  const phone = document.getElementById('c-phone').value.replace(/\D/g, '');
  if (cmode === 'pair' && (phone.length < 7 || phone.length > 15)) {
    errEl.textContent = '❌ Enter a valid number with country code';
    errEl.style.display = 'block'; return;
  }
  const btn = document.getElementById('c-start-btn');
  btn.disabled = true; btn.innerHTML = '<span class="spin"></span> Connecting...';
  try {
    const d = await fetch('/api/bot/start', {
      method: 'POST', headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ phoneNumber: phone, useQR: cmode === 'qr' })
    }).then(r => r.json());
    if (!d.success) {
      errEl.textContent = '❌ ' + (d.error || 'Failed');
      errEl.style.display = 'block';
      btn.disabled = false; btn.innerHTML = 'Get Started →'; return;
    }
    cSessId = d.sessionId;
    document.getElementById('c-s1').style.display = 'none';
    document.getElementById(cmode === 'pair' ? 'c-s2-pair' : 'c-s2-qr').style.display = 'block';

  } catch {
    errEl.textContent = '❌ Server error'; errEl.style.display = 'block';

  }
}

async function doFirebaseLogin() {
  const email = (document.getElementById('fb-email')?.value || '').trim();
  const pw    = (document.getElementById('fb-pw')?.value   || '').trim();
  const errEl = document.getElementById('fb-err');
  const btn   = document.getElementById('fb-btn');
  errEl.style.display = 'none';
  if (!email) { errEl.textContent = '❌ Enter your email address'; errEl.style.display = 'block'; return; }
  if (!pw)    { errEl.textContent = '❌ Enter your password';       errEl.style.display = 'block'; return; }
  btn.disabled = true; btn.textContent = '⏳ Verifying...';
  try {
    const r = await fetch('/api/admin/login', {
      method: 'POST', headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ email, password: pw })
    }).then(r => r.json());
    if (r.success && r.token) {
      adminToken = r.token;
      localStorage.setItem('adminToken', adminToken);
      showAdmin();
    } else {
      errEl.textContent = r.error || '❌ Login failed — check email & password';
      errEl.style.display = 'block';
      btn.disabled = false; btn.textContent = '🔓 Login to Admin Panel';
    }
  } catch {
    errEl.textContent = '❌ Server error — try again';
    errEl.style.display = 'block';
    btn.disabled = false; btn.textContent = '🔓 Login to Admin Panel';
  }
}


async function verifyToken() {
  try {
    const r = await fetch('/api/admin/verify', { headers: { 'x-admin-token': adminToken || '' } });
    if (r.status === 401) return false;
    const d = await r.json();
    return d.ok === true;
  } catch { return false; }
}

  localStorage.removeItem('adminToken'); adminToken = null;
  clearInterval(refreshTimer);
  showConnect();
}

async function apiGet(url) {
  const r = await fetch(url, { headers: { 'x-admin-token': adminToken || '' } });
  if (r.status === 401) { adminLogout(); throw new Error('Unauthorized'); }
  return r.json();
}
async function apiPost(url, data) {
  const r = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json', 'x-admin-token': adminToken || '' }, body: JSON.stringify(data) });
  if (r.status === 401) { adminLogout(); throw new Error('Unauthorized'); }
  return r.json();
}
async function apiDel(url) {
  return fetch(url, { method: 'DELETE', headers: { 'x-admin-token': adminToken || '' } }).then(r => r.json());
}

async function asStart() {
  const errEl = document.getElementById('as-err');
  errEl.style.display = 'none';
  const phone = document.getElementById('as-phone').value.replace(/\D/g, '');
  if (asmode === 'pair' && (phone.length < 7 || phone.length > 15)) { errEl.textContent = '❌ Valid number required'; errEl.style.display = 'block'; return; }
  const btn = document.getElementById('as-start-btn');
  btn.disabled = true; btn.innerHTML = '<span class="spin"></span>';
  try {
    const d = await fetch('/api/bot/start', {
      method: 'POST', headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ phoneNumber: phone, useQR: asmode === 'qr' })
    }).then(r => r.json());
    if (!d.success) { errEl.textContent = '❌ ' + (d.error || 'Failed'); errEl.style.display = 'block'; btn.disabled = false; btn.innerHTML = 'Start →'; return; }
    asSessId = d.sessionId;
    document.getElementById(asmode === 'pair' ? 'as-code-wrap' : 'as-qr-wrap').style.display = 'block';
    asPoll();
  } catch { errEl.textContent = '❌ Server error'; errEl.style.display = 'block'; btn.disabled = false; btn.innerHTML = 'Start →'; }
}
function asPoll() {
  clearInterval(asPollT);
  asPollT = setInterval(async () => {
    if (!asSessId) return;
    try {
      const d = await fetch('/api/bot/status/' + asSessId).then(r => r.json());
      if (d.pairingCode) document.getElementById('as-code').textContent = d.pairingCode;
      if (d.qrDataUrl) {
        document.getElementById('as-qr-ph').style.display = 'none';
        const img = document.getElementById('as-qr-img');
        img.src = d.qrDataUrl; img.style.display = 'block';
      }
      if (d.status === 'connected') {
        clearInterval(asPollT);
        document.getElementById('as-code-wrap').style.display = 'none';
        document.getElementById('as-qr-wrap').style.display   = 'none';
        document.getElementById('as-ok').style.display        = 'block';
        document.getElementById('as-start-btn').style.display = 'none';
        loadSessions();
      }
    } catch {}
  }, 1500);
}
function asCopy() {
  const txt = document.getElementById('as-code').textContent;
  if (!txt || txt === '----') return;
  navigator.clipboard.writeText(txt.replace(/-/g, '')).catch(() => {});
}

async function loadUsers() {
  document.getElementById('users-tbody').innerHTML = `<tr><td colspan="9" style="text-align:center;padding:20px;color:var(--muted)"><span class="spin"></span> Loading...</td></tr>`;
  try {
    const d = await apiGet('/api/admin/users');
    allUsers = d.users || []; filteredUsers = [...allUsers]; usersPage = 1; renderUsers();
  } catch { document.getElementById('users-tbody').innerHTML = `<tr><td colspan="9" style="text-align:center;padding:20px;color:var(--muted)">Failed to load</td></tr>`; }
}
function filterUsers() {
  const q = document.getElementById('user-search').value.toLowerCase();
  filteredUsers = q ? allUsers.filter(u => u.number.includes(q) || (u.name || '').toLowerCase().includes(q)) : [...allUsers];
  usersPage = 1; renderUsers();
}
function renderUsers() {
  const total = filteredUsers.length;
  document.getElementById('user-count').textContent = `${total} user${total !== 1 ? 's' : ''}`;
  const start = (usersPage - 1) * PER_PAGE;
  const page  = filteredUsers.slice(start, start + PER_PAGE);
  const tbody = document.getElementById('users-tbody');
  if (!page.length) { tbody.innerHTML = `<tr><td colspan="9" style="text-align:center;padding:24px;color:var(--muted)">No users found</td></tr>`; renderPages(0); return; }
  tbody.innerHTML = page.map((u, i) => `<tr>
    <td class="td-muted">${start + i + 1}</td>
    <td class="td-num">+${u.number}</td>
    <td class="td-name">${esc(u.name || 'Unknown')}</td>
    <td><span class="badge badge-blue">Lv ${u.level}</span></td>
    <td class="td-muted">${u.commandsUsed}</td>
    <td class="td-muted">💰 ${u.balance ?? 0}</td>
    <td class="td-muted" style="font-size:11px">${timeAgo(u.lastSeen)}</td>
    <td>${u.banned ? '<span class="badge badge-red">Banned</span>' : u.premium ? '<span class="badge badge-yellow">⭐ Premium</span>' : '<span class="badge badge-green">Active</span>'}</td>
    <td style="display:flex;gap:5px;flex-wrap:wrap">${u.banned
      ? `<button class="btn-unban" onclick="doUnban('${u.jid}')">Unban</button>`
      : `<button class="btn-ban" onclick="doBan('${u.jid}')">Ban</button>`}
      ${u.premium ? '' : `<button class="btn-unban" style="font-size:11px" onclick="doPremium('${u.jid}',true)">⭐</button>`}
      <button class="btn-unban" style="font-size:11px;background:#1e3a5f;color:#93c5fd" onclick="openHistory('${u.jid}','${esc(u.name||u.number)}')">📋</button>
      <button class="btn-ban" style="font-size:11px;border-color:rgba(239,68,68,.7)" onclick="doDeleteUser('${u.jid}')">🗑</button>
    </td>
  </tr>`).join('');
  renderPages(total);
}
function renderPages(total) {
  const pages = Math.ceil(total / PER_PAGE);
  const el = document.getElementById('users-pages');
  if (pages <= 1) { el.innerHTML = ''; return; }
  el.innerHTML = Array.from({ length: pages }, (_, i) => `<button class="page-btn${usersPage === i + 1 ? ' active' : ''}" onclick="goPage(${i + 1})">${i + 1}</button>`).join('');
}
function goPage(p) { usersPage = p; renderUsers(); }

async function doBan(jid) {
  if (!confirm(`Ban +${jid.split('@')[0]}?`)) return;
  await apiPost(`/api/admin/users/${encodeURIComponent(jid)}/ban`, {});
  const u = allUsers.find(x => x.jid === jid); if (u) u.banned = true; renderUsers();
}
async function doUnban(jid) {
  await apiPost(`/api/admin/users/${encodeURIComponent(jid)}/unban`, {});
  const u = allUsers.find(x => x.jid === jid); if (u) u.banned = false; renderUsers();
}
async function doDeleteUser(jid) {
  const num = '+' + jid.split('@')[0];
  if (!confirm(`Delete user ${num} and all their data? This cannot be undone.`)) return;
  try {
    const d = await apiDel(`/api/admin/users/${encodeURIComponent(jid)}`);
    if (d.success) {
      allUsers = allUsers.filter(x => x.jid !== jid);
      filteredUsers = filteredUsers.filter(x => x.jid !== jid);
      renderUsers();
    } else {
      alert('❌ Delete failed: ' + (d.error || 'Unknown error'));
    }
  } catch { alert('❌ Network error'); }
}

    msg.textContent = '❌ Failed to save settings';
    msg.className = 'settings-msg msg-err'; msg.style.display = 'block';
  }
}

// CHANGE ADMIN CREDENTIALS
// ════════════════════════════════════════════════════════════
async function changeAdminPassword() {
  const msg       = document.getElementById('cp-msg');
  const current   = document.getElementById('cp-current').value;
  const newEmail  = document.getElementById('cp-email').value.trim();
  const newPass   = document.getElementById('cp-new').value;
  const confirm   = document.getElementById('cp-confirm').value;
  msg.style.display = 'none';

  if (!current) {
    msg.textContent = '❌ Current password is required.';
    msg.className = 'settings-msg msg-err'; msg.style.display = 'block'; return;
  }
  if (!newPass && !newEmail) {
    msg.textContent = '❌ Enter a new password or new email (or both).';
    msg.className = 'settings-msg msg-err'; msg.style.display = 'block'; return;
  }
  if (newPass && newPass !== confirm) {
    msg.textContent = '❌ New passwords do not match.';
    msg.className = 'settings-msg msg-err'; msg.style.display = 'block'; return;
  }
  if (newPass && newPass.length < 6) {
    msg.textContent = '❌ New password must be at least 6 characters.';
    msg.className = 'settings-msg msg-err'; msg.style.display = 'block'; return;
  }

  msg.textContent = '⏳ Updating credentials in Firebase...';
  msg.className = 'settings-msg msg-ok'; msg.style.display = 'block';

  try {
    const d = await apiPost('/api/admin/change-password', {
      currentPassword: current,
      newEmail:        newEmail  || undefined,
      newPassword:     newPass   || undefined,
    });
    if (d.success) {
      msg.textContent = '✅ ' + (d.message || 'Credentials updated! Logging out...');
      msg.className = 'settings-msg msg-ok';
      ['cp-current','cp-email','cp-new','cp-confirm'].forEach(id => { document.getElementById(id).value = ''; });
      setTimeout(() => adminLogout(), 2500);
    } else {
      msg.textContent = d.error || '❌ Failed to update credentials.';
      msg.className = 'settings-msg msg-err';
    }
  } catch {
    msg.textContent = '❌ Network error — try again.';
    msg.className = 'settings-msg msg-err';
  }
}

// BROADCAST
// ════════════════════════════════════════════════════════════
async function loadBroadcast() {
  try {
    // Populate session dropdown
    const sel = document.getElementById('bc-session');
    if (sel) {
      const ss = await apiGet('/api/admin/sessions');
      // Keep first "auto" option, rebuild the rest
      while (sel.options.length > 1) sel.remove(1);
      (ss.sessions || []).forEach(s => {
        if (s.status === 'connected') {
          const opt = document.createElement('option');
          opt.value = s.sessionId;
          opt.textContent = (s.number ? '+' + s.number + ' — ' : '') + s.sessionId;
          sel.appendChild(opt);
        }
      });
    }
    // Show user count
    const users = await apiGet('/api/admin/users');
    const active = (users.users || []).filter(u => !u.banned);
    const cnt = document.getElementById('bc-count');
    if (cnt) cnt.textContent = active.length + ' user' + (active.length !== 1 ? 's' : '') + ' will receive this';
  } catch(e) {
    const cnt = document.getElementById('bc-count');
    if (cnt) cnt.textContent = '— could not load user count';
  }
}

async function sendBroadcast() {
  const msg   = document.getElementById('bc-msg');
  const btn   = document.getElementById('bc-btn');
  const res   = document.getElementById('bc-result');
  const sessEl = document.getElementById('bc-session');
  if (!msg || !msg.value.trim()) {
    res.textContent = '❌ Please type a message before sending.';
    res.className = 'settings-msg msg-err'; res.style.display = 'block';
    return;
  }
  btn.disabled = true; btn.textContent = '⏳ Sending...';
  res.style.display = 'none';
  try {
    const d = await apiPost('/api/admin/broadcast', {
      message: msg.value.trim(),
      sessionId: sessEl ? sessEl.value : '',
    });
    if (d.success) {
      res.textContent = `✅ Sent to ${d.sent} user${d.sent !== 1 ? 's' : ''} (${d.failed} failed, ${d.total} total)`;
      res.className = 'settings-msg msg-ok';
      msg.value = '';
    } else {
      res.textContent = '❌ ' + (d.error || 'Broadcast failed');
      res.className = 'settings-msg msg-err';
    }
    res.style.display = 'block';
  } catch(e) {
    res.textContent = '❌ Network error — is the bot connected?';
    res.className = 'settings-msg msg-err'; res.style.display = 'block';
  } finally {
    btn.disabled = false; btn.textContent = '📤 Send Broadcast';
    setTimeout(() => { res.style.display = 'none'; }, 8000);
  }
}

    count.textContent = d.error
      ? `⚠️ ${d.error}`
      : `${groups.length} group${groups.length !== 1 ? 's' : ''}`;

    </tr>`).join('');

    count.textContent = 'Error';
    tbody.innerHTML = `<tr><td colspan="5" style="text-align:center;padding:24px;color:#f87171">❌ Failed to load groups — is the bot connected?</td></tr>`;
  }
}

// ════════════════════════════════════════════════════════════
// HELPERS
// ════════════════════════════════════════════════════════════
function esc(s) { return String(s || '').replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;'); }
function fmtTime(ts) {
  if (!ts) return '--';
  return new Date(ts).toLocaleTimeString('en-PK', { hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false });
}
function fmtDate(ts) {
  if (!ts) return '--';
  return new Date(ts).toLocaleString('en-PK', { day:'2-digit', month:'short', hour:'2-digit', minute:'2-digit', hour12:true });
}
function timeAgo(ts) {
  if (!ts) return 'Never';
  const s = Math.floor((Date.now() - ts) / 1000);
  if (s < 60) return 'Just now';
  if (s < 3600) return Math.floor(s / 60) + 'm ago';
  if (s < 86400) return Math.floor(s / 3600) + 'h ago';
  return Math.floor(s / 86400) + 'd ago';
}
